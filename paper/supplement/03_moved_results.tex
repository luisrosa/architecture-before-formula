\section{Results moved from the main dependency spine}\label{supp:moved-results}

This section records exact results that remain useful but are not required to identify the minimal receiver-factorization object or derive the Transformer normal form.

\subsection{Information lost under a deterministic aggregate}

\begin{proposition}[Evidence accounting under a deterministic quotient]\label{prop:supp-information-loss}
Let $Y$ and $Z$ be finite-valued random variables and let $Z'=q(Z)$ be deterministic. Then
\[
\boxed{
I(Y;Z)=I(Y;Z')+I(Y;Z\mid Z').
}
\]
Equivalently,
\[
H(Y\mid Z')-H(Y\mid Z)=I(Y;Z\mid Z')\ge0.
\]
\end{proposition}

\begin{proof}
Because $Z'$ is a function of $Z$, the pair $(Z,Z')$ contains the same information as $Z$. The chain rule gives
\[
I(Y;Z,Z')=I(Y;Z')+I(Y;Z\mid Z'),
\]
and the left side is $I(Y;Z)$. The entropy form follows from the definition of conditional mutual information.
\end{proof}

Applied to a source-resolved presentation $P(H)$ and aggregate $E(H)=\Agg P(H)$, the conditional term measures target-relevant evidence present before aggregation and absent afterward.

\subsection{Signed and normalized presentations of an FFN}

Let
\[
\FFN(x)-b_2=\sum_ea_e(x)w_e.
\]

\begin{corollary}[Signed nonnegative presentation]\label{cor:supp-ffn-signed}
Define
\[
a_e^+=\max\{a_e,0\},
\qquad
a_e^-=\max\{-a_e,0\},
\]
and the doubled carrier $\widetilde E=(\{+\}\times E)\sqcup(\{-\}\times E)$ with payloads $v(+,e)=w_e$ and $v(-,e)=-w_e$. Then
\[
\FFN(x)-b_2
=
\sum_{(\sigma,e)\in\widetilde E}a_e^\sigma(x)v(\sigma,e)
\]
with nonnegative coefficients.
\end{corollary}

\begin{proof}
Use $a_e=a_e^+-a_e^-$ and sum coordinatewise.
\end{proof}

\begin{corollary}[Amplitude--allocation--direction form]\label{cor:supp-ffn-amplitude}
If
\[
A(x):=\sum_e(a_e^+(x)+a_e^-(x))>0,
\]
define $p_x(\sigma,e)=a_e^\sigma(x)/A(x)$. Then $p_x$ is a probability law and
\[
\FFN(x)-b_2
=
A(x)\sum_{\sigma,e}p_x(\sigma,e)v(\sigma,e).
\]
When $A(x)=0$, the routed contribution vanishes.
\end{corollary}

This construction separates total amplitude from a normalized hidden-direction allocation. Defining $-\tau\log p_x$ gives a tautological hidden potential on the positive support, but not the compositional-evidence result obtained for attention.

\subsection{Finite typed local approximation}

\begin{theorem}[Finite typed family approximation]\label{thm:supp-typed-approximation}
Let $T$ be finite. For each $t\in T$, let $K_t\subset\R^d$ be compact and $f_t:K_t\to\R^p$ continuous. Let $e_t\in\R^{|T|}$ be one-hot. For every $\varepsilon>0$, there exists one two-layer network $N:\R^{d+|T|}\to\R^p$ with continuous nonpolynomial activation such that
\[
\sup_{t\in T}\sup_{x\in K_t}\norm{N(x,e_t)-f_t(x)}<\varepsilon.
\]
\end{theorem}

\begin{proof}
The set $K=\bigcup_t(K_t\times\{e_t\})$ is compact. The function $f(x,e_t)=f_t(x)$ is continuous on this finite disjoint union. Universal approximation gives the result \citep{leshno1993multilayer}.
\end{proof}

Type marks can therefore condition one shared local continuation without being erased from the interface.

\subsection{Two-stage residual error propagation}

Let
\[
U=H+T(H),
\qquad
B(H)=U+F(U),
\]
and let
\[
\widehat U=H+\widehat T(H),
\qquad
\widehat B(H)=\widehat U+\widehat F(\widehat U).
\]

\begin{proposition}[Two-stage residual error propagation]\label{prop:supp-block-error-ledger}
Suppose
\[
\norm{T(H)-\widehat T(H)}\le\varepsilon_T,
\]
\[
\sup_{V\in D}\norm{F(V)-\widehat F(V)}\le\varepsilon_F,
\]
and $F$ is $L_F$-Lipschitz on a domain containing $U$ and $\widehat U$. Then
\[
\boxed{
\norm{B(H)-\widehat B(H)}
\le(1+L_F)\varepsilon_T+\varepsilon_F.
}
\]
\end{proposition}

\begin{proof}
First $\norm{U-\widehat U}\le\varepsilon_T$. Then
\[
\begin{aligned}
\norm{B(H)-\widehat B(H)}
&\le\norm{U-\widehat U}
+\norm{F(U)-\widehat F(\widehat U)}\\
&\le\varepsilon_T+L_F\varepsilon_T+\varepsilon_F.
\end{aligned}
\]
\end{proof}

The ledger separates approximation error in the routed interface from approximation error in the local continuation. It begins after the exact interface has been chosen and therefore does not remove an interface-fiber lower bound.
